cmake_minimum_required(VERSION 3.16)
project(WpsClockKylin VERSION 2.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt5 5.12 REQUIRED COMPONENTS Core Gui Widgets Multimedia)
find_package(X11 REQUIRED)
add_executable(WpsClock main.cpp x11probe.cpp)
target_include_directories(WpsClock PRIVATE ${X11_INCLUDE_DIR})
target_link_libraries(WpsClock PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets Qt5::Multimedia ${X11_LIBRARIES} ${X11_Xext_LIB})
target_compile_options(WpsClock PRIVATE -Wall -Wextra)
enable_testing()
add_executable(model-tests ../../tests/linux/model_tests.cpp)
add_test(NAME model COMMAND model-tests)
add_executable(appdata-tests ../../tests/linux/appdata_tests.cpp)
target_link_libraries(appdata-tests PRIVATE Qt5::Core Qt5::Gui)
add_test(NAME appdata COMMAND appdata-tests)
add_executable(x11-tests ../../tests/linux/x11_tests.cpp x11probe.cpp)
target_include_directories(x11-tests PRIVATE ${X11_INCLUDE_DIR})
target_link_libraries(x11-tests PRIVATE Qt5::Core Qt5::Gui ${X11_LIBRARIES} ${X11_Xext_LIB})
# Run x11-tests in a 1280x800 Xvfb + window-manager session with xdotool installed.
option(WPSCLOCK_X11_TESTS "Register tests for an isolated X11 desktop" OFF)
if(WPSCLOCK_X11_TESTS)
    add_test(NAME x11 COMMAND x11-tests)
    set_tests_properties(x11 PROPERTIES TIMEOUT 60)
endif()
